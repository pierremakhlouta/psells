# Runtime dependencies: what PSells needs in order to run.
#
# psells.py, the command line application, still uses only the standard library
# and needs nothing here. These are for the HTTP API in api.py and the web pages
# in web.py, which one uvicorn process serves together.
#
# Development and testing tools live in requirements-dev.txt, so a deployment
# installs what it needs to run and nothing else.
fastapi==0.141.1
uvicorn==0.53.0

# The template engine for the web pages. FastAPI does not install it, because
# not every FastAPI application renders HTML.
jinja2==3.1.6

# Reads the body of a form post. FastAPI checks for it when a route that takes
# form fields is defined, at import, so without it the whole application fails
# to start, the API included. Not the unrelated package called plain multipart.
python-multipart==0.0.32

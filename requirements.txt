# Runtime dependencies: what PSells needs in order to run.
#
# One image holds all of it: psells.py, the command line application, needs the
# PostgreSQL driver, and the HTTP API in api.py and the web pages in web.py,
# which one uvicorn process serves together, need the rest.
#
# Development and testing tools live in requirements-dev.txt, so a deployment
# installs what it needs to run and nothing else.
fastapi==0.143.5
uvicorn==0.54.0

# The template engine for the web pages. FastAPI does not install it, because
# not every FastAPI application renders HTML.
jinja2==3.1.6

# Reads the body of a form post. FastAPI checks for it when a route that takes
# form fields is defined, at import, so without it the whole application fails
# to start, the API included. Not the unrelated package called plain multipart.
python-multipart==0.0.32
# The PostgreSQL driver, used by psells.connect and so by every way in. [binary]
# brings a compiled copy of the C client library, so nothing has to be built.
psycopg[binary]==3.3.6


# Hashes and checks the login password with Argon2id, in auth.py. Brings
# argon2-cffi-bindings, the compiled Argon2 library, and cffi with it; both
# come as prebuilt wheels for the Mac and for the image.
argon2-cffi==25.1.0

# Reads an uploaded photo, turns it the right way up, shrinks it to 1600 px on
# its long side and writes it as a JPEG with no metadata, in photos.py. JPEG
# and PNG only, so no second image library for HEIC. 13.0.0 within a week of
# its release, against the usual wait, because it fixes some thirty flaws in
# 12.3.0, several reachable from Image.open on an uploaded file.
pillow==13.0.0
